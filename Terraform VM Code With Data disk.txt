# Resource Group
resource "azurerm_resource_group" "s1" {
  name     = "RG-01"
  location = "East US"
}

# Virtual Network
resource "azurerm_virtual_network" "s2" {
  name                = "vnet-01"
  location            = azurerm_resource_group.s1.location
  resource_group_name = azurerm_resource_group.s1.name
  address_space       = ["10.0.0.0/16"]
  depends_on          = [azurerm_resource_group.s1]
}

# Subnet
resource "azurerm_subnet" "s3" {
  name                 = "subnet-01"
  resource_group_name  = azurerm_resource_group.s1.name
  virtual_network_name = azurerm_virtual_network.s2.name
  address_prefixes     = ["10.0.1.0/24"]
  depends_on           = [azurerm_virtual_network.s2]
}

# Public IP
resource "azurerm_public_ip" "pip" {
  name                = "pip-vm-01"
  location            = azurerm_resource_group.s1.location
  resource_group_name = azurerm_resource_group.s1.name

  allocation_method = "Static"
  sku               = "Standard"
}

# Network Security Group
resource "azurerm_network_security_group" "s4" {
  name                = "nsg-01"
  location            = azurerm_resource_group.s1.location
  resource_group_name = azurerm_resource_group.s1.name

  security_rule {
    name                       = "Allow-RDP"
    priority                   = 100
    direction                  = "Inbound"
    access                     = "Allow"
    protocol                   = "Tcp"
    source_port_range          = "*"
    destination_port_range     = "3389"
    source_address_prefix      = "*"
    destination_address_prefix = "*"
  }
}

# Network Interface
resource "azurerm_network_interface" "s5" {
  name                = "nic-01"
  location            = azurerm_resource_group.s1.location
  resource_group_name = azurerm_resource_group.s1.name

  ip_configuration {
    name                          = "internal"
    subnet_id                     = azurerm_subnet.s3.id
    private_ip_address_allocation = "Dynamic"
    public_ip_address_id          = azurerm_public_ip.pip.id
  }
}

# Attach NSG to NIC
resource "azurerm_network_interface_security_group_association" "nsg_association" {
  network_interface_id      = azurerm_network_interface.s5.id
  network_security_group_id = azurerm_network_security_group.s4.id
}

# Windows VM
resource "azurerm_windows_virtual_machine" "vm" {
  name                = "vm-01"
  computer_name       = "my-compute"
  location            = azurerm_resource_group.s1.location
  resource_group_name = azurerm_resource_group.s1.name

  size           = "Standard_B2s"
  admin_username = "Nusadmin"
  admin_password = var.admin_password

  network_interface_ids = [
    azurerm_network_interface.s5.id
  ]

  os_disk {
    name                 = "osdisk-windows-vm"
    caching              = "ReadWrite"
    storage_account_type = "Standard_LRS"
  }

  source_image_reference {
    publisher = "MicrosoftWindowsServer"
    offer     = "WindowsServer"
    sku       = "2022-datacenter-azure-edition"
    version   = "latest"
  }
}

# Managed Data Disk
resource "azurerm_managed_disk" "data_disk" {
  name                = "datadisk-windows-vm-01"
  location            = azurerm_resource_group.s1.location
  resource_group_name = azurerm_resource_group.s1.name

  storage_account_type = "Standard_LRS"
  create_option        = "Empty"
  disk_size_gb         = 32
}

# Attach Data Disk to VM
resource "azurerm_virtual_machine_data_disk_attachment" "data_disk_01" {
  managed_disk_id    = azurerm_managed_disk.data_disk.id
  virtual_machine_id = azurerm_windows_virtual_machine.vm.id

  lun     = 0
  caching = "ReadWrite"
}

# Admin Password Variable
variable "admin_password" {
  description = "Administrator password for Windows VM"
  type        = string
  sensitive   = true
}

# Output Public IP
output "public_ip_address" {
  value = azurerm_public_ip.pip.ip_address
}

# Output VM Name
output "vm_name" {
  value = azurerm_windows_virtual_machine.vm.name
}
